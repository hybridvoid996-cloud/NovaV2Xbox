function searchTokens(q){return [...new Set(norm(q).split(" ").filter(x=>x.length>1&&!["the","a","an","by","and","or","feat","ft"].includes(x)))]}
function searchWordMatches(token,word){
  if(word===token||(token.length>=4&&word.startsWith(token)))return true;
  if(token.length<5||Math.abs(token.length-word.length)>1)return false;
  let i=0,j=0,edits=0;
  while(i<token.length&&j<word.length){
    if(token[i]===word[j]){i++;j++;continue}
    if(++edits>1)return false;
    if(token.length>=word.length)i++;
    if(word.length>=token.length)j++;
  }
  return edits+(token.length-i)+(word.length-j)<=1;
}

const lycache=new Map();
function parseLrc(txt){
  const out=[];
  const offset=Number(String(txt||"").match(/\[offset:([+-]?\d+)\]/i)?.[1]||0)/1000;
  String(txt||"").split(/\r?\n/).forEach(l=>{
    const tags=[...l.matchAll(/\[(\d+):(\d+(?:\.\d+)?)\]/g)];
    const text=l.replace(/\[[^\]]*\]/g,"").trim();
    tags.forEach(m=>out.push({t:+m[1]*60+ +m[2]-offset,s:text}));
  });
  return out.filter(x=>x.s||true).sort((a,b)=>a.t-b.t);
}
function parseYrc(txt){
  const lines=[];let invalid=false;
  String(txt||"").split(/\r?\n/).forEach(row=>{
    const header=/^\[(\d+),(\d+)\]/.exec(row);
    if(!header)return;
    const start=+header[1]/1000,end=(+header[1]+ +header[2])/1000;
    const rawParts=[...row.slice(header[0].length).matchAll(/\((\d+),(\d+),\d+\)(.*?)(?=\(\d+,\d+,\d+\)|$)/g)]
      .map(m=>({t:+m[1]/1000,end:(+m[1]+ +m[2])/1000,s:m[3]}));
    if(!rawParts.length||rawParts.some((p,i)=>p.end<p.t||p.t<start-.1||p.end>end+.1||(i&&p.t<rawParts[i-1].t))){invalid=true;return}
    const parts=[];let prefix="";
    rawParts.forEach(p=>{
      if(p.s.trim()){p.s=prefix+p.s;prefix="";parts.push(p)}
      else if(parts.length)parts[parts.length-1].s+=p.s;
      else prefix+=p.s;
    });
    parts.forEach((p,i)=>{if(p.end===p.t)p.end=Math.max(p.t,Math.min(end,parts[i+1]?.t??end))});
    const s=parts.map(p=>p.s).join("").trim();
    lines.push(s?{t:start,s,parts}:{t:start,s:""});
  });
  if(invalid){console.warn("Invalid word-timed lyrics; using line-synced lyrics instead.");return[]}
  return lines.sort((a,b)=>a.t-b.t);
}

const CJK=/[\u2e80-\u303f\u3040-\u30ff\u3100-\u312f\u3400-\u4dbf\u4e00-\u9fff\uac00-\ud7af\uf900-\ufaff\uff00-\uffef]/;

function cleanLines(lines){
  const real=lines.filter(x=>x.s);
  const latin=real.filter(x=>/[A-Za-z]{2,}/.test(x.s)).length;
  if(!real.length||latin<real.length*.35)return lines;

  lines.forEach(x=>{

  });
  return out;
}
function wordLyricsMatchLines(words,reference){

function tryUrl(url,tok,start,want,timeoutMs=9000){
  return new Promise(resolve=>{
    let source;
    try{source=new URL(url,location.href).href}catch(error){console.warn("Invalid music source URL:",error);resolve("error");return}
    let doneFlag=false;
    const done=status=>{
      if(doneFlag)return;doneFlag=true;clearTimeout(timer);
      audio.removeEventListener("playing",playing);audio.removeEventListener("error",failed);
      audio.removeEventListener("loadedmetadata",metadata);
      if(cancelUrlTrial===cancel)cancelUrlTrial=null;
      resolve(status);
    };
    const current=()=>tok===playTok&&audio.src===source;
    const playing=()=>{if(current())done("playing")};
    const failed=()=>{if(current())done("error")};
    const metadata=()=>{
      if(!current())return;
      if(!Number.isFinite(audio.duration)||(want?.dur&&Math.abs(audio.duration-want.dur)>want.tol)){
        audio.pause();done("duration-mismatch");return;
      }
      if(start>0)audio.currentTime=Math.min(start,audio.duration);
      audio.play().catch(error=>{
        if(!current())return;
        console.warn("Music playback failed:",error);
        done(error.name==="NotAllowedError"?"blocked":"error");
      });
    };
    const cancel=()=>done("cancelled");
    const timer=setTimeout(()=>done("timeout"),timeoutMs);
    cancelUrlTrial=cancel;
    audio.pause();audio.removeAttribute("src");audio.load();
    audio.addEventListener("playing",playing);audio.addEventListener("error",failed);
    audio.addEventListener("loadedmetadata",metadata);audio.preload="metadata";audio.src=url;audio.load();
  });
}

function slideLyricView(el,p,instant=false){
  const win=el.ownerDocument.defaultView;
  cancelLyricScroll(el);
  const from=el.scrollTop;
  const to=p?Math.max(0,Math.min(el.scrollHeight-el.clientHeight,p.offsetTop-el.clientHeight/2+p.offsetHeight/2)):0;
  if(instant){lyricScrolls.delete(el);el.scrollTo({top:to,behavior:"instant"});return}
  if(Math.abs(to-from)<1){lyricScrolls.delete(el);return}
  const scroll={win,frame:0};let start;
  const step=now=>{
    if(start===undefined)start=now;
    const progress=Math.min(1,(now-start)/700);
    const eased=1-Math.pow(1-progress,3);
    el.scrollTo({top:from+(to-from)*eased,behavior:"instant"});
    if(progress<1)scroll.frame=win.requestAnimationFrame(step);
    else lyricScrolls.delete(el);
  };
  lyricScrolls.set(el,scroll);
  scroll.frame=win.requestAnimationFrame(step);
}

function card(o){return `<div class="card" data-k="${o.k}" data-id="${esc(o.id)}"><div class="im ${o.round?"r":""}"><img loading="lazy" src="${esc(o.img)}" alt="">${o.play===false?"":`<div class="pb">${IC.play}</div>`}</div><div class="n">${esc(o.name)}</div><div class="d">${esc(o.sub||"")}</div></div>`}
function bindCards(root){root.querySelectorAll(".card").forEach(c=>{c.onclick=e=>{const k=c.dataset.k,id=c.dataset.id;
  if(e.target.closest(".pb")){playCard(k,id);return}
  openCard(k,id)}})}
async function playCard(k,id){
  try{
   if(k==="recent"){const t=recent.find(x=>String(x.id)===String(id));if(t)playOne(t);return}
   if(k==="album"){const a=await dz("album/"+id);const tr=(a.tracks.data||[]).map(x=>T(x,a)).filter(Boolean);playQueue(tr,0)}
   else if(k==="pl"){const p=await dz("playlist/"+id+"?limit=100");playQueue(TL(p.tracks.data),0)}
   else if(k==="artist"){const tr=await artistTop(+id,30);playQueue(tr,0,{radio:tr[0]})}
   else if(k==="liked"){if(liked.length)playQueue(liked,0);else toast("No liked songs yet")}
   else if(k==="upl"){const p=playlists.find(x=>x.id===id);if(p?.tracks.length)playQueue(p.tracks,0);else toast("This playlist is empty")}
   else if(k==="track"){const t=await dz("track/"+id);playOne(T(t))}
  }catch{toast("Couldn't load that")}
}
function openCard(k,id){if(k==="recent"){playCard(k,id);return}if(k==="album")openAlbum(+id);else if(k==="pl")openPlaylist(+id);else if(k==="artist")openArtist(+id);else if(k==="liked")showView({name:"liked"});else if(k==="upl")showView({name:"upl",id});else if(k==="track")dz("track/"+id).then(t=>{t=T(t);playOne(t)})}
function trackRows(tracks,opts={}){
  return `<div class="tl"><div class="th"><span>#</span><span>Title</span><span>Album</span><span></span><span style="text-align:right">⏱</span></div>${tracks.map((t,i)=>trow(t,i)).join("")}</div>`;
}
function trow(t,i){return `<div class="tr" data-id="${esc(t.id)}" data-i="${i}"><div class="ix"><b>${i+1}</b>${IC.play}</div><div class="ti"><img loading="lazy" src="${esc(t.cover)}" alt=""><div style="min-width:0"><div class="tn">${t.explicit?'<span class="ex">E</span>':""}${esc(t.title)}</div><div class="ta"><span ${t.artistId?`data-aid="${esc(t.artistId)}"`:""}>${esc(t.artist)}</span></div></div></div><div class="al" ${t.albumId?`data-alb="${esc(t.albumId)}"`:""}>${esc(t.album)}</div><button class="hb ${isLiked(t.id)?"on":""}" data-like>${isLiked(t.id)?IC.heart:IC.heartO}</button><div class="du">${fmt(t.dur)}</div></div>`}
function bindRows(root,tracks,ctxFn){
  root.querySelectorAll(".tr").forEach(r=>{const t=tracks.find(x=>String(x.id)===r.dataset.id);if(!t)return;
    r.onclick=e=>{if(e.target.closest("[data-aid],[data-alb],[data-like]"))return;(ctxFn?ctxFn(t):playOne(t,tracks))};
    r.oncontextmenu=e=>{e.preventDefault();trackMenu(t,e.clientX,e.clientY)};
    r.querySelector("[data-like]").onclick=e=>{e.stopPropagation();toggleLike(t);r.querySelector("[data-like]").classList.toggle("on",isLiked(t.id));r.querySelector("[data-like]").innerHTML=isLiked(t.id)?IC.heart:IC.heartO};

    const al=r.querySelector("[data-alb]");al&&(al.onclick=e=>{e.stopPropagation();openAlbum(t.albumId)});
  });
  $$(".tr[data-id]").forEach(r=>r.classList.toggle("cur",!!P.cur()&&String(P.cur().id)===r.dataset.id));
}

let sT,suggestGeneration=0,suggestionIndex=-1,searchSuggestions=[];

function closeSearchDropdown(){
  ++suggestGeneration;
  searchDropdown.hidden=true;
  searchDropdown.classList.remove("open");
  searchInput.setAttribute("aria-expanded","false");
  suggestionIndex=-1;
}
function playSuggestion(index){
  const t=searchSuggestions[index];
  if(!t)return false;
  closeSearchDropdown();
  playQueue([t],0,{radio:t.artistId?t:null});
  return true;
}
async function updateSearchDropdown(q){
  const generation=++suggestGeneration;
  searchInput.setAttribute("aria-expanded","true");
  searchDropdown.hidden=false;
  searchDropdown.classList.remove("open");
  void searchDropdown.offsetWidth;
  searchDropdown.classList.add("open");
  searchDropdown.innerHTML=`<div class="search-drop-empty">Searching…</div>`;
  try{
    const [songResult,artistResult]=await Promise.allSettled([
      dzCached(`search?q=${encodeURIComponent(q)}&limit=30`).then(result=>TL(result.data)),
      dzCached(`search/artist?q=${encodeURIComponent(q)}&limit=20`).then(result=>rankSearchArtists(q,result.data||[]))
    ]);
    if(songResult.status==="rejected")console.warn("Song suggestions unavailable:",songResult.reason);
    if(generation!==suggestGeneration||searchInput.value.trim()!==q)return;
    if(artistResult.status==="rejected")console.warn("Artist suggestions unavailable:",artistResult.reason);
    const artist=artistResult.status==="fulfilled"?artistResult.value[0]:null;
    const artistSongs=artist?await artistTop(artist.id,7):[];
    if(generation!==suggestGeneration||searchInput.value.trim()!==q)return;
    const ranked=rankSearchTracks(q,songResult.status==="fulfilled"?songResult.value:[]);
    const seen=new Set();
    searchSuggestions=[...artistSongs,...ranked].filter(t=>{const key=String(t.id);if(seen.has(key))return false;seen.add(key);return true}).slice(0,7);
    if(!searchSuggestions.length){
      try{searchSuggestions=rankSearchTracks(q,await ytmSearch(q,20)).slice(0,7)}
      catch(error){console.warn("YouTube Music suggestions unavailable:",error)}
      if(generation!==suggestGeneration||searchInput.value.trim()!==q)return;
    }
    if(!searchSuggestions.length&&!artist){
      searchDropdown.innerHTML=`<div class="search-drop-empty">No close matches for “${esc(q)}”. Press Enter to search all results.</div>`;
      return;
    }
    searchDropdown.innerHTML=`<div class="search-drop-head"><span>Results</span><button type="button" data-all>See all results</button></div>${artist?`<button class="search-suggestion" type="button" data-artist="${esc(artist.id)}"><img src="${esc(artist.picture_medium||"")}" alt=""><span class="txt"><b>${esc(artist.name)}</b><small>Artist</small></span></button>`:""}${
      searchSuggestions.map((t,i)=>`<button class="search-suggestion" type="button" role="option" aria-selected="false" data-suggestion="${i}" style="animation-delay:${i*24}ms"><img src="${esc(t.cover)}" alt=""><span class="txt"><b>${esc(t.title)}</b><small>${esc(t.artist)}</small></span><span class="playmark">${IC.play}</span></button>`).join("")
    }`;
    searchDropdown.querySelector("[data-all]").onclick=()=>{closeSearchDropdown();showView({name:"search",q})};
    const artistButton=searchDropdown.querySelector("[data-artist]");
    if(artistButton)artistButton.onclick=()=>{closeSearchDropdown();openArtist(+artist.id)};
    searchDropdown.querySelectorAll("[data-suggestion]").forEach(button=>{
      button.onclick=()=>playSuggestion(Number(button.dataset.suggestion));
    });
  }catch(error){
    if(generation!==suggestGeneration)return;
    console.warn("Music search suggestions failed:",error);
    searchDropdown.innerHTML=`<div class="search-drop-empty">Search unavailable. Press Enter to try the full search.</div>`;
  }
}
function setSuggestionIndex(index){
  const options=[...searchDropdown.querySelectorAll("[data-suggestion]")];
  suggestionIndex=Math.max(0,Math.min(index,options.length-1));
  options.forEach((option,i)=>{
    option.classList.toggle("active",i===suggestionIndex);
    option.setAttribute("aria-selected",String(i===suggestionIndex));
  });
  options[suggestionIndex]?.scrollIntoView({block:"nearest"});
}
