function accentTextFor(hex){
  const r=parseInt(hex.slice(1,3),16)/255,g=parseInt(hex.slice(3,5),16)/255,b=parseInt(hex.slice(5,7),16)/255;
  const lin=c=>c<=0.03928?c/12.92:Math.pow((c+0.055)/1.055,2.4);
  const lum=0.2126*lin(r)+0.7152*lin(g)+0.0722*lin(b);
  return lum>0.5?'#05060a':'#ffffff';
}

let musicArtSequence=0;
function applyArtColor(url){
  const sequence=++musicArtSequence;
  if(!url){return}
  const img=new Image();
  img.crossOrigin='anonymous';
  img.onload=()=>{
    if(sequence!==musicArtSequence)return;
    try{
      const c=document.createElement('canvas');const s=24;
      c.width=s;c.height=s;const ctx=c.getContext('2d');
      ctx.drawImage(img,0,0,s,s);
      const d=ctx.getImageData(0,0,s,s).data;
      const colors=extractPalette(d);
      document.documentElement.style.setProperty('--art-1','rgb('+colors[0].join(',')+')');
      document.documentElement.style.setProperty('--art-2','rgb('+colors[1].join(',')+')');
    }catch(e){}
  };
  img.onerror=()=>{};
  img.src=url;
}
function extractPalette(d){
  const buckets={};
  for(let i=0;i<d.length;i+=4){
    if(d[i+3]<128)continue;
    const r=d[i]>>4<<4,g=d[i+1]>>4<<4,b=d[i+2]>>4<<4;
    const key=r+','+g+','+b;buckets[key]=(buckets[key]||0)+1;
  }
  const sorted=Object.entries(buckets).sort((a,b)=>b[1]-a[1]);
  const top=sorted.slice(0,3).map(([k])=>k.split(',').map(Number));
  if(!top.length)return[[34,211,238],[129,140,248]];
  while(top.length<2)top.push([34,211,238]);
  if(Math.abs(top[0][0]-top[1][0])+Math.abs(top[0][1]-top[1][1])+Math.abs(top[0][2]-top[1][2])<60)top[1]=[129,140,248];
  return top;
}

function drawVisualizer(){
  const canvases=[{el:$('visBar'),bars:18,gap:2},{el:$('visFs'),bars:64,gap:3}];
  const t=performance.now()/1000;
  const active=state.isPlaying?1:0;
  for(const cfg of canvases){
    const cv=cfg.el;if(!cv)continue;
    const ctx=cv.getContext('2d');if(!ctx)continue;
    const w=cv.width,h=cv.height;
    ctx.clearRect(0,0,w,h);
    const n=cfg.bars;
    const bw=(w-(n-1)*cfg.gap)/n;
    for(let i=0;i<n;i++){
      const idx=Math.floor(i/n*visData.length);
      let v;
      if(active){
        const freq=(i+1)/n;
        const base=Math.sin(t*6+i*0.5)*0.5+0.5;
        const base2=Math.sin(t*3.3+i*0.9+1)*0.5+0.5;
        const noise=Math.random()*0.25;
        v=(base*0.5+base2*0.3+noise)*(1.1-freq*0.5);
        v=Math.max(0.06,v);
      }else{v=0.06+Math.sin(t*1.5+i)*0.02}
      visData[idx]=visData[idx]*0.6+v*0.4;
      const bh=Math.max(cfg.gap,visData[idx]*h);
      const x=i*(bw+cfg.gap);
      const grad=ctx.createLinearGradient(0,h,0,h-bh);
      grad.addColorStop(0,'rgba('+accentRgb[0]+','+accentRgb[1]+','+accentRgb[2]+',0.3)');
      grad.addColorStop(1,accentColor);
      ctx.fillStyle=grad;
      const r=Math.min(bw/2,3);
      roundRect(ctx,x,h-bh,bw,bh,r);ctx.fill();
    }
  }
  requestAnimationFrame(drawVisualizer);
}
function roundRect(ctx,x,y,w,h,r){
  r=Math.min(r,w/2,h/2);ctx.beginPath();ctx.moveTo(x+r,y);
  ctx.arcTo(x+w,y,x+w,y+h,r);ctx.arcTo(x+w,y+h,x,y+h,r);
  ctx.arcTo(x,y+h,x,y,r);ctx.arcTo(x,y,x+w,y,r);ctx.closePath();
}
function probeAccent(){

      document.body.appendChild(probe);const out=getComputedStyle(probe).color;document.body.removeChild(probe);

      }
    }
  }catch(e){}
}
